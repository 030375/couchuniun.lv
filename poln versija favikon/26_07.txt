<script>
// Установите дату начала курса здесь (год, месяц-1, день)
const courseStartDate = new Date(2025, 10, 15); // 1 сентября 2025

function updateCountdown() {
    const now = new Date();
    const diff = courseStartDate - now;
    
    // Если дата наступила
    if (diff <= 0) {
        document.getElementById('countdown-timer').innerHTML = `
            <div class="col-span-4 text-center py-4">
                <div class="text-2xl font-bold text-primary">Курс начался!</div>
            </div>
        `;
        return;
    }
    
    // Рассчет временных единиц
    const days = Math.floor(diff / (1000 * 60 * 60 * 24));
    const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
    const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
    const seconds = Math.floor((diff % (1000 * 60)) / 1000);
    
    // Обновление элементов
    document.getElementById('countdown-days').textContent = days.toString().padStart(2, '0');
    document.getElementById('countdown-hours').textContent = hours.toString().padStart(2, '0');
    document.getElementById('countdown-minutes').textContent = minutes.toString().padStart(2, '0');
    document.getElementById('countdown-seconds').textContent = seconds.toString().padStart(2, '0');
}

// Первый запуск
updateCountdown();

// Обновление каждую секунду
setInterval(updateCountdown, 1000);
</script>
