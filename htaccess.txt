# Optional Apache .htaccess for https://coachunion.lv
# Use this file only on Apache hosting. If your hosting is Nginx, configure these rules in the server panel instead.

DirectoryIndex index.html
Options -Indexes

<IfModule mod_rewrite.c>
RewriteEngine On

# Force HTTPS and non-www canonical domain
RewriteCond %{HTTPS} !=on [OR]
RewriteCond %{HTTP_HOST} ^www\.coachunion\.lv$ [NC]
RewriteRule ^ https://coachunion.lv%{REQUEST_URI} [R=301,L]
</IfModule>

# Correct MIME types
AddType application/xml .xml
AddType text/plain .txt
AddType application/manifest+json .webmanifest

# Browser cache
<IfModule mod_expires.c>
ExpiresActive On
ExpiresByType text/css "access plus 30 days"
ExpiresByType application/javascript "access plus 30 days"
ExpiresByType image/jpeg "access plus 90 days"
ExpiresByType image/png "access plus 90 days"
ExpiresByType image/svg+xml "access plus 90 days"
ExpiresByType image/x-icon "access plus 90 days"
ExpiresByType application/pdf "access plus 30 days"
</IfModule>

# Compression
<IfModule mod_deflate.c>
AddOutputFilterByType DEFLATE text/html text/plain text/xml text/css application/javascript application/json application/xml image/svg+xml
</IfModule>

# Basic security headers
<IfModule mod_headers.c>
Header always set X-Content-Type-Options "nosniff"
Header always set X-Frame-Options "SAMEORIGIN"
Header always set Referrer-Policy "strict-origin-when-cross-origin"
Header always set Permissions-Policy "geolocation=(), microphone=(), camera=()"
</IfModule>

ErrorDocument 404 /404.html
